# opg-infra — Repo, Cloudflare Infrastructure, Provider Router, Deploy

export async function handleHealth(_req: Request, env: Env): Promise<Response> {
  let db = 'unknown';
  try {
    const r = await env.OPG_DB.prepare('SELECT 1 AS ok').first<{ ok: number }>();
    db = r.ok === 1 ? 'ok' : 'degraded';
  } catch {

}

- [ ] **Step 2:** Write `providers/Router.ts`:

```ts
import type { ProviderAdapter, ProviderResult } from './types';

export class ProviderRouter {
  private adapters: Map<string, ProviderAdapter> = new Map();

  register(adapter: ProviderAdapter): void {
    this.adapters.set(adapter.id, adapter);
  }

  async test(providerId: string, apiKey: string): Promise<ProviderResult> {
    const adapter = this.adapters.get(providerId);
    if (!adapter) return { ok: false, provider: providerId, error: 'unknown provider' };
    return adapter.test(apiKey);
  }

  list(): { id: string; label: string }[] {
    return [...this.adapters.values()].map(a => ({ id: a.id, label: a.label }));
  }
}
```
